{{- /* Body image render hook — post-images redline. Treatment is a property of the image's class, decided by nb_image_class.html: cover.class-style override via {class="..."} on the markdown image, else default-by-extension. Only a photograph may crop (never in the body — that's the aspect-preserving .cmyk .print construction, opt into a wider {bleed=true} track), everything else prints untreated at min(intrinsic, track width), never upscaled. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path }} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- $ext := strings.TrimPrefix "." (path.Ext $u.Path) -}} {{- $override := index .Attributes "class" -}} {{- $class := partial "nb_image_class.html" (dict "override" $override "ext" $ext) -}} {{- $bleed := and (eq $class "photograph") (eq (index .Attributes "bleed") "true") -}} {{- $caption := "" -}} {{- with .Title }}{{ $caption = . }}{{- else }}{{- with $.Text }}{{ $caption = . }}{{- end }}{{- end -}} {{- $figClass := "nb-figure" -}} {{- if eq $class "mark" }}{{ $figClass = printf "%s nb-figure-mark" $figClass }}{{ else if $bleed }}{{ $figClass = printf "%s nb-figure-bleed" $figClass }}{{ end -}} {{- if eq $class "photograph" }} {{- else if eq $class "screenshot" }} {{- else }} {{- end -}}
